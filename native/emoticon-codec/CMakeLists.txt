cmake_minimum_required(VERSION 3.16.3)
cmake_policy(SET CMP0091 NEW)
project(werc_emoticon_codec LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(flag ENABLE_SIMD ENABLE_AVX2 ENABLE_AVX512 ENABLE_DECODER ENABLE_ENCODER ENABLE_SDL ENABLE_SHERLOCK265 ENABLE_INTERNAL_DEVELOPMENT_TOOLS)
 set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(source/libde265-1.1.3 de265)
add_library(jbig STATIC source/jbigkit-2.2/libjbig/jbig.c source/jbigkit-2.2/libjbig/jbig_ar.c)
target_include_directories(jbig PUBLIC source/jbigkit-2.2/libjbig)
add_executable(werc-emoticon-codec codec.cpp)
target_compile_definitions(werc-emoticon-codec PRIVATE LIBDE265_STATIC_BUILD)
target_include_directories(werc-emoticon-codec PRIVATE source/libde265-1.1.3 ${CMAKE_CURRENT_BINARY_DIR}/de265)
target_link_libraries(werc-emoticon-codec PRIVATE de265 jbig)
if(MSVC)
 target_compile_options(werc-emoticon-codec PRIVATE /utf-8 /W4)
 target_link_options(werc-emoticon-codec PRIVATE /DYNAMICBASE /NXCOMPAT)
endif()
